<script>
    let tg = window.Telegram.WebApp;
    tg.expand(); // جعل النافذة تفتح بملء الشاشة تلقائياً داخل تيليجرام

    function submitData() {
        let name = document.getElementById("fullname").value.trim();
        let phone = document.getElementById("phone").value.trim();
        let address = document.getElementById("address").value.trim();
        let purpose = document.getElementById("purpose").value;

        if(!name || !phone || !address) {
            alert("يرجى تعبئة جميع الحقول المطلوبة بدقة!");
            return;
        }

        // تنسيق البيانات لتُرسل للبوت دفعة واحدة
        let dataString = `• الاسم: ${name}\n• الهاتف: ${phone}\n• العنوان: ${address}\n• الغرض: ${purpose}`;

        // إرسال البيانات للبوت وإغلاق نافذة الويب المصغرة فوراً
        tg.sendData(dataString);
        tg.close();
    }
</script>
